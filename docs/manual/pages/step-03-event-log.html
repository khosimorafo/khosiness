What this step does

Record run facts as typed events in an append-only JSON Lines file, then load them back in order.

What it builds on

Steps 1 and 2 give us task identity and model response shapes. They do not yet provide a durable account of what happened.

Goal

Preserve a factual history for inspection and later recovery without treating that whole history as model context.

End state

Event and JsonlEventStore support newline-terminated append, flush, ordered reload, and explicit failure on malformed lines. The loop will emit real run events in Step 4.

src/khosiness/event_store.py complete file 
from pathlib import Path

from .events import Event


class JsonlEventStore:
    """Append-only JSON Lines event store."""

    def __init__(self, path: Path):
        self.path = path
        self.path.parent.mkdir(parents=True, exist_ok=True)

    def append(self, event: Event) -> None:
        with self.path.open("a", encoding="utf-8") as handle:
            handle.write(event.model_dump_json() + "\n")
            handle.flush()

    def load_all(self) -> list[Event]:
        if not self.path.exists():
            return []

        events: list[Event] = []
        with self.path.open("r", encoding="utf-8") as handle:
            for line in handle:
                line = line.strip()
                if line:
                    events.append(Event.model_validate_json(line))
        return events

from khosiness.event_store import JsonlEventStore
from khosiness.events import Event


def test_empty_store_loads_as_empty_list(tmp_path):
    store = JsonlEventStore(tmp_path / "events.jsonl")
    assert store.load_all() == []


def test_event_store_preserves_order(tmp_path):
    store = JsonlEventStore(tmp_path / "events.jsonl")

    store.append(Event(run_id="r1", type="first"))
    store.append(Event(run_id="r1", type="second"))

    assert [event.type for event in store.load_all()] == [
        "first",
        "second",
    ]


def test_event_payload_round_trips(tmp_path):
    store = JsonlEventStore(tmp_path / "events.jsonl")
    store.append(
        Event(
            run_id="r1",
            type="tool_completed",
            payload={"tool": "read_file", "ok": True},
        )
    )

    restored = store.load_all()[0]

    assert restored.payload == {
        "tool": "read_file",
        "ok": True,
    }


def test_many_events_round_trip(tmp_path):
    store = JsonlEventStore(tmp_path / "events.jsonl")

    for index in range(1000):
        store.append(
            Event(
                run_id="r1",
                type="tick",
                payload={"index": index},
            )
        )

    events = store.load_all()

    assert len(events) == 1000
    assert events[-1].payload["index"] == 999
