What this step does

Coordinate one bounded turn at a time: build context, call the model, handle a proposed tool call or final text, update state, and record events.

What it builds on

The Task sets the limit; the model adapter returns proposals; the event store preserves facts. Step 4 connects those separate boundaries.

Goal

Make each transition observable and stop when a final answer or step limit is reached, without an endless hidden conversation.

End state

RunState and a minimal run loop pass tests for a two-turn tool-then-answer path, a first-turn answer, a step limit, and a returned fake-tool failure. Real tools and policy come later.

Tests to perform

  1. Tool call then final answer completes in two turns.
  2. Final answer on first turn works.
  3. max_steps stops an unfinished run.
  4. A tool error becomes an observation rather than crashing the loop.
Deliberate-failure rule: after the happy path works, deliberately trigger at least one failure and inspect the returned result, state change and event trace.

from collections.abc import Callable

from .event_store import JsonlEventStore
from .events import Event
from .state import RunState
from .task import Task


def run(
    *,
    task: Task,
    state: RunState,
    model,
    context_builder,
    tools,
    events: JsonlEventStore,
    save_state: Callable[[RunState], None] | None = None,
) -> RunState:
    """Run the smallest genuine observe -> decide -> act loop."""

    save_state = save_state or (lambda _: None)

    while not state.finished and state.step < task.max_steps:
        state.step += 1
        messages = context_builder.build(task, state)

        events.append(
            Event(
                run_id=state.run_id,
                type="model_called",
                payload={"step": state.step},
            )
        )

        response = model.generate(
            messages=messages,
            tools=tools.schemas(),
        )

        events.append(
            Event(
                run_id=state.run_id,
                type="model_responded",
                payload={
                    "step": state.step,
                    "has_text": response.text is not None,
                    "tool_call_count": len(response.tool_calls),
                },
            )
        )

        if response.tool_calls:
            for call in response.tool_calls:
                events.append(
                    Event(
                        run_id=state.run_id,
                        type="tool_requested",
                        payload={
                            "name": call.name,
                            "arguments": call.arguments,
                        },
                    )
                )

                result = tools.execute(call)

                state.observations.append(
                    {
                        "tool": call.name,
                        "arguments": call.arguments,
                        "result": result.model_dump(),
                    }
                )

                events.append(
                    Event(
                        run_id=state.run_id,
                        type=(
                            "tool_completed"
                            if result.ok
                            else "tool_failed"
                        ),
                        payload={
                            "name": call.name,
                            "ok": result.ok,
                        },
                    )
                )

        elif response.text is not None:
            state.finished = True
            state.final_answer = response.text
            events.append(
                Event(
                    run_id=state.run_id,
                    type="task_completed",
                    payload={"step": state.step},
                )
            )

        save_state(state)

    if not state.finished and state.step >= task.max_steps:
        events.append(
            Event(
                run_id=state.run_id,
                type="step_limit_reached",
                payload={"max_steps": task.max_steps},
            )
        )

    return state

from pathlib import Path

from khosiness.event_store import JsonlEventStore
from khosiness.loop import run
from khosiness.models.base import ModelResponse, ToolCall
from khosiness.models.fake import FakeModel
from khosiness.state import RunState
from khosiness.task import Task
from khosiness.tools.base import ToolResult


class StaticContext:
    def build(self, task, state):
        return [
            {
                "role": "user",
                "content": task.objective,
            }
        ]


class FakeTools:
    def __init__(self):
        self.calls = []

    def schemas(self):
        return [{"name": "echo"}]

    def execute(self, call):
        self.calls.append(call)
        return ToolResult(
            ok=True,
            content=call.arguments["text"],
        )


def make_task(tmp_path: Path, max_steps: int = 5):
    return Task(
        id="task-1",
        objective="Use echo, then finish",
        workspace=tmp_path,
        max_steps=max_steps,
    )


def test_loop_executes_tool_then_finishes(tmp_path):
    fake_model = FakeModel(
        [
            ModelResponse(
                tool_calls=[
                    ToolCall(
                        id="1",
                        name="echo",
                        arguments={"text": "hello"},
                    )
                ]
            ),
            ModelResponse(text="finished"),
        ]
    )
    tools = FakeTools()
    events = JsonlEventStore(tmp_path / "events.jsonl")
    state = RunState(run_id="run-1")

    result = run(
        task=make_task(tmp_path),
        state=state,
        model=fake_model,
        context_builder=StaticContext(),
        tools=tools,
        events=events,
    )

    assert result.finished is True
    assert result.final_answer == "finished"
    assert result.step == 2
    assert len(tools.calls) == 1
    assert result.observations[0]["result"]["content"] == "hello"


def test_loop_stops_on_first_final_answer(tmp_path):
    result = run(
        task=make_task(tmp_path),
        state=RunState(run_id="run-1"),
        model=FakeModel([ModelResponse(text="done")]),
        context_builder=StaticContext(),
        tools=FakeTools(),
        events=JsonlEventStore(tmp_path / "events.jsonl"),
    )

    assert result.step == 1
    assert result.final_answer == "done"


def test_loop_obeys_step_limit(tmp_path):
    fake_model = FakeModel(
        [
            ModelResponse(
                tool_calls=[
                    ToolCall(
                        id=str(index),
                        name="echo",
                        arguments={"text": "again"},
                    )
                ]
            )
            for index in range(5)
        ]
    )

    result = run(
        task=make_task(tmp_path, max_steps=2),
        state=RunState(run_id="run-1"),
        model=fake_model,
        context_builder=StaticContext(),
        tools=FakeTools(),
        events=JsonlEventStore(tmp_path / "events.jsonl"),
    )

    assert result.finished is False
    assert result.step == 2
