What this step does

Introduce named tool contracts, a registry, workspace path checks, and structured results for read-only filesystem operations.

What it builds on

The Step 4 reference dispatches injected fake tools and records their results. Step 5 supplies concrete tools through that same interface.

Goal

Make a request resolve to a known operation, reject path escapes before the file read, and return usable evidence or an explicit failure.

Starter implementation

Executable reference boundaries: Pydantic argument models generate schemas and validate each filesystem request before I/O. Read ranges are 1-based and inclusive. Reads reject files over 1,000,000 bytes; returned content is capped at 20,000 characters; directory lists and searches stop at 100 entries/matches. A truncated result is incomplete evidence. Directory/search order is not guaranteed. Search checks each resolved target, including symlink files. These checks assume a stable local workspace: they do not prevent concurrent path replacement, bound total traversal time, or implement an OS sandbox. Those execution controls belong to later stages. Risk labels remain metadata; approval policy arrives in Step 9.
Do not paste blindly. Type this in VS Code, make it run, then explain every line before you add another layer.
from pydantic import BaseModel, Field
from abc import ABC, abstractmethod

class ToolResult(BaseModel):
    ok: bool
    content: str
    metadata: dict = Field(default_factory=dict)

class Tool(ABC):
    name: str
    description: str
    risk: str = "read_only"

    @abstractmethod
    def schema(self) -> dict: ...

    @abstractmethod
    def execute(self, arguments: dict) -> ToolResult: ...

Tests to perform

  1. Unknown tool gives structured error.
  2. Path traversal outside workspace is denied.
  3. Missing file gives structured error.
  4. Search output is bounded.
  5. Line-range reads return only requested lines.
  6. Malformed arguments fail before filesystem access.
  7. Search rejects discovered symlink escapes before reading them.
  8. Directory, search, read-output, and file-byte limits are tested.
Deliberate-failure rule: after the happy path works, deliberately trigger at least one failure and inspect the returned result, state change and event trace.

from khosiness.models.base import ToolCall

from .base import Tool, ToolResult


class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, Tool] = {}

    def register(self, tool: Tool) -> None:
        if tool.name in self._tools:
            raise ValueError(f"duplicate tool: {tool.name}")
        self._tools[tool.name] = tool

    def schemas(self) -> list[dict]:
        return [tool.schema() for tool in self._tools.values()]

    def execute(self, call: ToolCall) -> ToolResult:
        tool = self._tools.get(call.name)

        if tool is None:
            return ToolResult(
                ok=False,
                content=f"Unknown tool: {call.name}",
            )

        try:
            return tool.execute(call.arguments)
        except Exception as exc:
            return ToolResult(
                ok=False,
                content=f"{type(exc).__name__}: {exc}",
            )
src/khosiness/tools/filesystem.py complete file 
from itertools import islice
from pathlib import Path

from pydantic import BaseModel, ConfigDict, Field, model_validator

from .base import Tool, ToolResult


MAX_FILE_BYTES = 1_000_000
MAX_OUTPUT_CHARS = 20_000
MAX_ENTRIES = 100
MAX_MATCHES = 100


class PathArguments(BaseModel):
    model_config = ConfigDict(strict=True, extra="forbid")
    path: str = Field(min_length=1)


class ReadArguments(PathArguments):
    start_line: int = Field(default=1, ge=1)
    end_line: int | None = Field(default=None, ge=1)

    @model_validator(mode="after")
    def ordered_range(self):
        if self.end_line is not None and self.end_line < self.start_line:
            raise ValueError("end_line must be at least start_line")
        return self


class SearchArguments(PathArguments):
    query: str = Field(min_length=1)


def safe_path(workspace: Path, relative_path: str) -> Path:
    workspace = workspace.resolve()
    candidate = (workspace / relative_path).resolve()
    if not candidate.is_relative_to(workspace):
        raise ValueError("path escapes workspace")
    return candidate


def read_text_bounded(path: Path) -> str:
    if not path.is_file():
        raise ValueError(f"not a file: {path.name}")
    with path.open("rb") as stream:
        data = stream.read(MAX_FILE_BYTES + 1)
    if len(data) > MAX_FILE_BYTES:
        raise ValueError("file exceeds byte limit")
    return data.decode("utf-8")


def schema_for(tool: Tool, arguments: type[BaseModel]) -> dict:
    return {
        "name": tool.name,
        "description": tool.description,
        "parameters": arguments.model_json_schema(),
    }


class ListDirectoryTool(Tool):
    name = "list_directory"
    description = "List at most 100 entries inside the workspace."

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return schema_for(self, PathArguments)

    def execute(self, arguments: dict) -> ToolResult:
        args = PathArguments.model_validate(arguments)
        path = safe_path(self.workspace, args.path)
        if not path.is_dir():
            raise ValueError(f"not a directory: {args.path}")
        # Select a bounded subset, then sort it. Directory order is unspecified.
        items = list(islice(path.iterdir(), MAX_ENTRIES + 1))
        entries = sorted(
            item.name + ("/" if not item.is_symlink() and item.is_dir() else "")
            for item in items[:MAX_ENTRIES]
        )
        text = "\n".join(entries)
        return ToolResult(
            ok=True,
            content=text[:MAX_OUTPUT_CHARS],
            metadata={
                "count": len(entries),
                "truncated": len(items) > MAX_ENTRIES or len(text) > MAX_OUTPUT_CHARS,
            },
        )


class ReadFileTool(Tool):
    name = "read_file"
    description = "Read a UTF-8 file; optional line bounds are 1-based and inclusive."

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return schema_for(self, ReadArguments)

    def execute(self, arguments: dict) -> ToolResult:
        args = ReadArguments.model_validate(arguments)
        path = safe_path(self.workspace, args.path)
        text = read_text_bounded(path)
        selected = "".join(text.splitlines(keepends=True)[args.start_line - 1:args.end_line])
        return ToolResult(
            ok=True,
            content=selected[:MAX_OUTPUT_CHARS],
            metadata={
                "characters": len(text),
                "start_line": args.start_line,
                "end_line": args.end_line,
                "truncated": len(selected) > MAX_OUTPUT_CHARS,
            },
        )


class SearchFilesTool(Tool):
    name = "search_files"
    description = "Search UTF-8 files; return at most 100 matches and 20,000 characters."

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return schema_for(self, SearchArguments)

    def execute(self, arguments: dict) -> ToolResult:
        args = SearchArguments.model_validate(arguments)
        root = safe_path(self.workspace, args.path)
        if not root.is_dir():
            raise ValueError(f"not a directory: {args.path}")
        matches: list[str] = []
        output_chars = 0
        for path in root.rglob("*"):
            # Validate every discovered target, including symlink files.
            resolved = safe_path(self.workspace, str(path.relative_to(self.workspace)))
            if not resolved.is_file():
                continue
            try:
                lines = read_text_bounded(resolved).splitlines()
            except UnicodeDecodeError:
                continue
            for number, line in enumerate(lines, start=1):
                if args.query in line:
                    match = f"{path.relative_to(self.workspace)}:{number}:{line.strip()}"
                    matches.append(match)
                    output_chars += len(match) + (1 if len(matches) > 1 else 0)
                    if len(matches) >= MAX_MATCHES or output_chars >= MAX_OUTPUT_CHARS:
                        return ToolResult(
                            ok=True,
                            content="\n".join(matches)[:MAX_OUTPUT_CHARS],
                            metadata={"truncated": True},
                        )
        return ToolResult(ok=True, content="\n".join(matches), metadata={"truncated": False})

from khosiness.models.base import ToolCall
from khosiness.tools.filesystem import (
    ListDirectoryTool,
    ReadFileTool,
    SearchFilesTool,
)
from khosiness.tools.registry import ToolRegistry


def test_registry_rejects_duplicate_tool(tmp_path):
    registry = ToolRegistry()
    tool = ReadFileTool(tmp_path)
    registry.register(tool)

    try:
        registry.register(tool)
        assert False, "expected duplicate registration to fail"
    except ValueError:
        pass


def test_unknown_tool_returns_structured_error():
    registry = ToolRegistry()

    result = registry.execute(
        ToolCall(
            id="1",
            name="missing",
            arguments={},
        )
    )

    assert result.ok is False
    assert "Unknown tool" in result.content


def test_read_file(tmp_path):
    (tmp_path / "hello.txt").write_text(
        "hello",
        encoding="utf-8",
    )
    tool = ReadFileTool(tmp_path)

    result = tool.execute({"path": "hello.txt"})

    assert result.ok is True
    assert result.content == "hello"


def test_read_file_cannot_escape_workspace(tmp_path):
    tool = ReadFileTool(tmp_path)

    try:
        tool.execute({"path": "../outside.txt"})
        assert False, "expected path escape to fail"
    except ValueError as exc:
        assert "escapes workspace" in str(exc)


def test_list_directory(tmp_path):
    (tmp_path / "a.txt").write_text("a", encoding="utf-8")
    (tmp_path / "folder").mkdir()

    result = ListDirectoryTool(tmp_path).execute({"path": "."})

    assert result.content.splitlines() == [
        "a.txt",
        "folder/",
    ]


def test_search_files(tmp_path):
    (tmp_path / "config.py").write_text(
        "RETRY_COUNT = 3\n",
        encoding="utf-8",
    )

    result = SearchFilesTool(tmp_path).execute(
        {"query": "RETRY", "path": "."}
    )

    assert "config.py:1:RETRY_COUNT = 3" in result.content

def test_missing_file_returns_structured_error(tmp_path):
    registry = ToolRegistry()
    registry.register(ReadFileTool(tmp_path))
    result = registry.execute(ToolCall(id="1", name="read_file", arguments={"path": "missing.txt"}))
    assert result.ok is False
    assert "not a file" in result.content


def test_argument_validation_precedes_io(tmp_path):
    registry = ToolRegistry()
    registry.register(ReadFileTool(tmp_path))
    for arguments in [{}, {"path": 7}, {"path": "a", "extra": True},
                      {"path": "a", "start_line": 0}, {"path": "a", "start_line": "1"},
                      {"path": "a", "start_line": 3, "end_line": 2}]:
        result = registry.execute(ToolCall(id="1", name="read_file", arguments=arguments))
        assert result.ok is False
        assert "ValidationError" in result.content


def test_read_line_range(tmp_path):
    (tmp_path / "lines.txt").write_text("one\ntwo\nthree\nfour\n", encoding="utf-8")
    result = ReadFileTool(tmp_path).execute({"path": "lines.txt", "start_line": 2, "end_line": 3})
    assert result.content == "two\nthree\n"
    assert result.metadata["truncated"] is False


def test_read_output_and_byte_limits(tmp_path):
    from khosiness.tools.filesystem import MAX_FILE_BYTES, MAX_OUTPUT_CHARS
    path = tmp_path / "large.txt"
    path.write_text("x" * (MAX_OUTPUT_CHARS + 1), encoding="utf-8")
    result = ReadFileTool(tmp_path).execute({"path": path.name})
    assert len(result.content) == MAX_OUTPUT_CHARS
    assert result.metadata["truncated"] is True
    path.write_bytes(b"x" * (MAX_FILE_BYTES + 1))
    registry = ToolRegistry()
    registry.register(ReadFileTool(tmp_path))
    result = registry.execute(ToolCall(id="1", name="read_file", arguments={"path": path.name}))
    assert result.ok is False
    assert "byte limit" in result.content


def test_search_match_limit(tmp_path):

    result = SearchFilesTool(tmp_path).execute({"path": ".", "query": "RETRY"})
    assert len(result.content.splitlines()) == 100
    assert result.metadata["truncated"] is True


def test_search_character_limit(tmp_path):
