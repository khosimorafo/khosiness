from khosiness.models.base import ToolCall

from .base import Tool, ToolResult


class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, Tool] = {}

    def register(self, tool: Tool) -> None:
        if tool.name in self._tools:
            raise ValueError(f"duplicate tool: {tool.name}")
        self._tools[tool.name] = tool

    def schemas(self) -> list[dict]:
        return [tool.schema() for tool in self._tools.values()]

    def execute(self, call: ToolCall) -> ToolResult:
        tool = self._tools.get(call.name)

        if tool is None:
            return ToolResult(
                ok=False,
                content=f"Unknown tool: {call.name}",
            )

        try:
            return tool.execute(call.arguments)
        except Exception as exc:
            return ToolResult(
                ok=False,
                content=f"{type(exc).__name__}: {exc}",
            )

from pathlib import Path

from .base import Tool, ToolResult


def safe_path(workspace: Path, relative_path: str) -> Path:
    workspace = workspace.resolve()
    candidate = (workspace / relative_path).resolve()

    if not candidate.is_relative_to(workspace):
        raise ValueError("path escapes workspace")

    return candidate


class ListDirectoryTool(Tool):
    name = "list_directory"
    description = "List files and directories inside the workspace."

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "path": {"type": "string"},
                },
                "required": ["path"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        path = safe_path(self.workspace, arguments["path"])
        if not path.is_dir():
            raise ValueError(f"not a directory: {arguments['path']}")

        entries = sorted(
            item.name + ("/" if item.is_dir() else "")
            for item in path.iterdir()
        )
        return ToolResult(
            ok=True,
            content="\n".join(entries),
            metadata={"count": len(entries)},
        )


class ReadFileTool(Tool):
    name = "read_file"
    description = "Read a UTF-8 text file inside the workspace."

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "path": {"type": "string"},
                },
                "required": ["path"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        path = safe_path(self.workspace, arguments["path"])
        if not path.is_file():
            raise ValueError(f"not a file: {arguments['path']}")

        text = path.read_text(encoding="utf-8")
        return ToolResult(
            ok=True,
            content=text[:20_000],
            metadata={
                "characters": len(text),
                "truncated": len(text) > 20_000,
            },
        )


class SearchFilesTool(Tool):
    name = "search_files"
    description = "Search UTF-8 text files inside the workspace."

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "query": {"type": "string"},
                    "path": {"type": "string"},
                },
                "required": ["query", "path"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        root = safe_path(self.workspace, arguments["path"])
        query = arguments["query"]
        matches: list[str] = []

        for path in root.rglob("*"):
            if not path.is_file():
                continue
            try:
                lines = path.read_text(
                    encoding="utf-8"
                ).splitlines()
            except UnicodeDecodeError:
                continue

            for line_number, line in enumerate(lines, start=1):
                if query in line:
                    relative = path.relative_to(self.workspace)
                    matches.append(
                        f"{relative}:{line_number}:{line.strip()}"
                    )
                    if len(matches) >= 100:
                        return ToolResult(
                            ok=True,
                            content="\n".join(matches),
                            metadata={"truncated": True},
                        )

        return ToolResult(
            ok=True,
            content="\n".join(matches),
            metadata={"truncated": False},
        )

from khosiness.models.base import ToolCall
from khosiness.tools.filesystem import (
    ListDirectoryTool,
    ReadFileTool,
    SearchFilesTool,
)
from khosiness.tools.registry import ToolRegistry


def test_registry_rejects_duplicate_tool(tmp_path):
    registry = ToolRegistry()
    tool = ReadFileTool(tmp_path)
    registry.register(tool)

    try:
        registry.register(tool)
        assert False, "expected duplicate registration to fail"
    except ValueError:
        pass


def test_unknown_tool_returns_structured_error():
    registry = ToolRegistry()

    result = registry.execute(
        ToolCall(
            id="1",
            name="missing",
            arguments={},
        )
    )

    assert result.ok is False
    assert "Unknown tool" in result.content


def test_read_file(tmp_path):
    (tmp_path / "hello.txt").write_text(
        "hello",
        encoding="utf-8",
    )
    tool = ReadFileTool(tmp_path)

    result = tool.execute({"path": "hello.txt"})

    assert result.ok is True
    assert result.content == "hello"


def test_read_file_cannot_escape_workspace(tmp_path):
    tool = ReadFileTool(tmp_path)

    try:
        tool.execute({"path": "../outside.txt"})
        assert False, "expected path escape to fail"
    except ValueError as exc:
        assert "escapes workspace" in str(exc)


def test_list_directory(tmp_path):
    (tmp_path / "a.txt").write_text("a", encoding="utf-8")
    (tmp_path / "folder").mkdir()

    result = ListDirectoryTool(tmp_path).execute({"path": "."})

    assert result.content.splitlines() == [
        "a.txt",
        "folder/",
    ]


def test_search_files(tmp_path):
    (tmp_path / "config.py").write_text(
        "RETRY_COUNT = 3\n",
        encoding="utf-8",
    )

    result = SearchFilesTool(tmp_path).execute(
        {"query": "RETRY", "path": "."}
    )

    assert "config.py:1:RETRY_COUNT = 3" in result.content
