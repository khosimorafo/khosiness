from dataclasses import dataclass

from khosiness.state import RunState
from khosiness.task import Task


@dataclass
class ContextStats:
    messages: int
    characters: int


class ContextBuilder:
    """Select what the model sees for the next decision."""

    def __init__(
        self,
        max_recent_observations: int = 8,
        max_observation_chars: int = 8_000,
    ):
        self.max_recent_observations = max_recent_observations
        self.max_observation_chars = max_observation_chars
        self.last_stats = ContextStats(messages=0, characters=0)

    def build(
        self,
        task: Task,
        state: RunState,
    ) -> list[dict]:
        recent = state.observations[
            -self.max_recent_observations:
        ]

        rendered_observations = []
        for observation in recent:
            text = repr(observation)
            if len(text) > self.max_observation_chars:
                text = (
                    text[: self.max_observation_chars]
                    + "... [truncated]"
                )
            rendered_observations.append(text)

        messages = [
            {
                "role": "system",
                "content": (
                    "You are operating inside a controlled harness. "
                    "Use tools to gather evidence. "
                    "Never invent tool results."
                ),
            },
            {
                "role": "user",
                "content": (
                    f"Objective: {task.objective}\n"
                    f"Constraints: {task.constraints}\n"
                    f"Success criteria: {task.success_criteria}\n"
                    f"Current step: {state.step}"
                ),
            },
            {
                "role": "user",
                "content": (
                    "Recent observations:\n"
                    + "\n".join(rendered_observations)
                ),
            },
        ]

        self.last_stats = ContextStats(
            messages=len(messages),
            characters=sum(
                len(message["content"])
                for message in messages
            ),
        )

        return messages
tests/test_context_builder.py complete file 
from khosiness.context.builder import ContextBuilder
from khosiness.state import RunState
from khosiness.task import Task


def make_task(tmp_path):
    return Task(
        id="t1",
        objective="Find retry configuration",
        workspace=tmp_path,
        constraints=["Do not modify files"],
        success_criteria=["Name the defining file"],
    )


def test_context_always_contains_task_and_constraints(tmp_path):
    builder = ContextBuilder()
    messages = builder.build(
        make_task(tmp_path),
        RunState(run_id="r1"),
    )

    joined = "\n".join(
        message["content"] for message in messages
    )

    assert "Find retry configuration" in joined
    assert "Do not modify files" in joined


def test_context_keeps_only_recent_observations(tmp_path):
    state = RunState(
        run_id="r1",
        observations=[
            {"value": "old"},
            {"value": "middle"},
            {"value": "recent"},
        ],
    )
    builder = ContextBuilder(max_recent_observations=2)

    messages = builder.build(make_task(tmp_path), state)
    joined = "\n".join(
        message["content"] for message in messages
    )

    assert "old" not in joined
    assert "middle" in joined
    assert "recent" in joined


def test_large_observation_is_truncated(tmp_path):
    state = RunState(
        run_id="r1",
        observations=[{"value": "x" * 1000}],
    )
    builder = ContextBuilder(max_observation_chars=100)

    messages = builder.build(make_task(tmp_path), state)

    assert "[truncated]" in messages[-1]["content"]


def test_context_stats_are_recorded(tmp_path):
    builder = ContextBuilder()
    messages = builder.build(
        make_task(tmp_path),
        RunState(run_id="r1"),
    )

    assert builder.last_stats.messages == len(messages)
    assert builder.last_stats.characters > 0
