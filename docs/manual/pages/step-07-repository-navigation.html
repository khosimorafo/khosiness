from pathlib import Path

from .base import Tool, ToolResult


def safe_path(workspace: Path, relative_path: str) -> Path:
    workspace = workspace.resolve()
    candidate = (workspace / relative_path).resolve()

    if not candidate.is_relative_to(workspace):
        raise ValueError("path escapes workspace")

    return candidate


class ListDirectoryTool(Tool):
    name = "list_directory"
    description = (
        "List files and directories inside the workspace. "
        "Use shallow listings first."
    )

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "path": {"type": "string"},
                    "max_items": {
                        "type": "integer",
                        "minimum": 1,
                        "maximum": 500,
                    },
                },
                "required": ["path"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        path = safe_path(self.workspace, arguments["path"])
        max_items = min(
            int(arguments.get("max_items", 200)),
            500,
        )

        if not path.is_dir():
            raise ValueError(f"not a directory: {arguments['path']}")

        all_entries = sorted(
            item.name + ("/" if item.is_dir() else "")
            for item in path.iterdir()
        )
        selected = all_entries[:max_items]

        return ToolResult(
            ok=True,
            content="\n".join(selected),
            metadata={
                "count": len(selected),
                "truncated": len(all_entries) > len(selected),
            },
        )


class ReadFileTool(Tool):
    name = "read_file"
    description = (
        "Read a bounded line range from a UTF-8 text file."
    )

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "path": {"type": "string"},
                    "start_line": {
                        "type": "integer",
                        "minimum": 1,
                    },
                    "end_line": {
                        "type": "integer",
                        "minimum": 1,
                    },
                },
                "required": ["path"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        path = safe_path(self.workspace, arguments["path"])

        if not path.is_file():
            raise ValueError(f"not a file: {arguments['path']}")

        lines = path.read_text(
            encoding="utf-8"
        ).splitlines()

        start = max(int(arguments.get("start_line", 1)), 1)
        requested_end = int(
            arguments.get("end_line", start + 199)
        )
        end = min(requested_end, start + 199, len(lines))

        numbered = [
            f"{line_number:>5} | {lines[line_number - 1]}"
            for line_number in range(start, end + 1)
        ]

        return ToolResult(
            ok=True,
            content="\n".join(numbered),
            metadata={
                "path": str(path.relative_to(self.workspace)),
                "start_line": start,
                "end_line": end,
                "total_lines": len(lines),
            },
        )


class SearchFilesTool(Tool):
    name = "search_files"
    description = (
        "Search text files and return matching file paths, "
        "line numbers and matching lines."
    )

    def __init__(self, workspace: Path):
        self.workspace = workspace.resolve()

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "query": {"type": "string"},
                    "path": {"type": "string"},
                    "max_matches": {
                        "type": "integer",
                        "minimum": 1,
                        "maximum": 200,
                    },
                },
                "required": ["query", "path"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        root = safe_path(self.workspace, arguments["path"])
        query = arguments["query"]
        max_matches = min(
            int(arguments.get("max_matches", 100)),
            200,
        )
        matches: list[dict] = []

        for path in root.rglob("*"):
            if not path.is_file():
                continue

            try:
                lines = path.read_text(
                    encoding="utf-8"
                ).splitlines()
            except (UnicodeDecodeError, OSError):
                continue

            for line_number, line in enumerate(lines, start=1):
                if query in line:
                    matches.append(
                        {
                            "path": str(
                                path.relative_to(self.workspace)
                            ),
                            "line": line_number,
                            "text": line.strip(),
                        }
                    )

                    if len(matches) >= max_matches:
                        break

            if len(matches) >= max_matches:
                break

        rendered = "\n".join(
            f"{match['path']}:{match['line']}:{match['text']}"
            for match in matches
        )

        return ToolResult(
            ok=True,
            content=rendered,
            metadata={
                "matches": matches,
                "truncated": len(matches) >= max_matches,
            },
        )

from khosiness.tools.filesystem import (
    ListDirectoryTool,
    ReadFileTool,
    SearchFilesTool,
)


def build_sample_repo(tmp_path):
    (tmp_path / "src" / "db").mkdir(parents=True)
    (tmp_path / "src" / "service").mkdir(parents=True)

    (tmp_path / "src" / "db" / "config.py").write_text(
        "DB_RETRY_COUNT = 3\n",
        encoding="utf-8",
    )
    (tmp_path / "src" / "db" / "session.py").write_text(
        "def create_session():\n"
        "    return object()\n",
        encoding="utf-8",
    )
    (tmp_path / "src" / "service" / "worker.py").write_text(
        "from src.db.session import create_session\n\n"
        "def run():\n"
        "    return create_session()\n",
        encoding="utf-8",
    )


def test_search_returns_file_line_and_text(tmp_path):
    build_sample_repo(tmp_path)

    result = SearchFilesTool(tmp_path).execute(
        {
            "query": "DB_RETRY_COUNT",
            "path": ".",
        }
    )

    match = result.metadata["matches"][0]
    assert match["path"] == "src/db/config.py"
    assert match["line"] == 1
    assert match["text"] == "DB_RETRY_COUNT = 3"


def test_read_file_returns_line_numbers(tmp_path):
    build_sample_repo(tmp_path)

    result = ReadFileTool(tmp_path).execute(
        {
            "path": "src/service/worker.py",
            "start_line": 1,
            "end_line": 4,
        }
    )

    assert "    1 | from src.db.session" in result.content
    assert "    4 |     return create_session()" in result.content


def test_directory_listing_is_bounded(tmp_path):
    for index in range(10):
        (tmp_path / f"{index}.txt").write_text(
            "x",
            encoding="utf-8",
        )

    result = ListDirectoryTool(tmp_path).execute(
        {
            "path": ".",
            "max_items": 3,
        }
    )

    assert len(result.content.splitlines()) == 3
    assert result.metadata["truncated"] is True
