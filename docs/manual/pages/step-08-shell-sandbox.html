import subprocess, time

def run_command(argv: list[str], cwd, timeout: int = 20):
    started = time.perf_counter()
    proc = subprocess.run(
        argv, cwd=cwd, text=True, capture_output=True,
        timeout=timeout, shell=False,
        env={"PATH": "/usr/bin:/bin"},
    )
    return {
        "exit_code": proc.returncode,
        "stdout": proc.stdout[-20000:],
        "stderr": proc.stderr[-20000:],
        "duration_ms": int((time.perf_counter()-started)*1000),
    }

Tests to perform

  1. Capture stdout.
  2. Nonzero exit becomes data.
  3. Timeout kills a long command.
  4. Unknown/forbidden command is denied.
  5. The real working tree remains untouched in sandbox tests.
Deliberate-failure rule: after the happy path works, deliberately trigger at least one failure and inspect the returned result, state change and event trace.

import os
import subprocess
import time
from pathlib import Path

from .base import ExecRequest, ExecResult


class LocalSandbox:
    """A constrained local runner used before introducing containers.

    It deliberately:
    - rejects commands outside an allowlist;
    - constrains cwd to the workspace;
    - uses shell=False;
    - passes a minimal environment;
    - bounds output;
    - enforces a timeout.
    """

    def __init__(
        self,
        workspace: Path,
        allowed_commands: set[str] | None = None,
    ):
        self.workspace = workspace.resolve()
        self.allowed_commands = allowed_commands or {
            "python",
            "pytest",
            "git",
            "ruff",
            "mypy",
            "grep",
            "find",
        }

    def execute(self, request: ExecRequest) -> ExecResult:
        if not request.argv:
            raise ValueError("argv must not be empty")

        command = Path(request.argv[0]).name
        if command not in self.allowed_commands:
            raise PermissionError(
                f"command is not allowed: {command}"
            )

        cwd = request.cwd.resolve()
        if not cwd.is_relative_to(self.workspace):
            raise PermissionError("cwd escapes workspace")

        started = time.perf_counter()

        try:
            process = subprocess.run(
                request.argv,
                cwd=cwd,
                text=True,
                capture_output=True,
                timeout=request.timeout_seconds,
                shell=False,
                env={
                    "PATH": os.environ.get("PATH", ""),
                    "PYTHONUNBUFFERED": "1",
                },
            )
            return ExecResult(
                exit_code=process.returncode,
                stdout=process.stdout[-20_000:],
                stderr=process.stderr[-20_000:],
                duration_ms=int(
                    (time.perf_counter() - started) * 1000
                ),
            )
        except subprocess.TimeoutExpired as exc:
            return ExecResult(
                exit_code=None,
                stdout=(exc.stdout or "")[-20_000:]
                if isinstance(exc.stdout, str)
                else "",
                stderr=(exc.stderr or "")[-20_000:]
                if isinstance(exc.stderr, str)
                else "",
                duration_ms=int(
                    (time.perf_counter() - started) * 1000
                ),
                timed_out=True,
            )

from pathlib import Path

from khosiness.sandbox.base import ExecRequest
from khosiness.sandbox.local import LocalSandbox

from .base import Tool, ToolResult


class ShellTool(Tool):
    name = "shell"
    description = (
        "Execute an allowed development command inside "
        "the controlled workspace runtime."
    )
    risk = "process_execution"

    def __init__(
        self,
        workspace: Path,
        sandbox: LocalSandbox,
    ):
        self.workspace = workspace.resolve()
        self.sandbox = sandbox

    def schema(self) -> dict:
        return {
            "name": self.name,
            "description": self.description,
            "parameters": {
                "type": "object",
                "properties": {
                    "argv": {
                        "type": "array",
                        "items": {"type": "string"},
                    },
                    "cwd": {"type": "string"},
                    "timeout_seconds": {
                        "type": "integer",
                        "minimum": 1,
                        "maximum": 120,
                    },
                },
                "required": ["argv"],
            },
        }

    def execute(self, arguments: dict) -> ToolResult:
        cwd = (
            self.workspace / arguments.get("cwd", ".")
        ).resolve()

        result = self.sandbox.execute(
            ExecRequest(
                argv=arguments["argv"],
                cwd=cwd,
                timeout_seconds=int(
                    arguments.get("timeout_seconds", 20)
                ),
            )
        )

        content = (
            f"exit_code={result.exit_code}\n"
            f"timed_out={result.timed_out}\n"
            f"stdout:\n{result.stdout}\n"
            f"stderr:\n{result.stderr}"
        )

        return ToolResult(
            ok=(
                not result.timed_out
                and result.exit_code == 0
            ),
            content=content,
            metadata=result.model_dump(),
        )

import sys
from pathlib import Path

from khosiness.sandbox.base import ExecRequest
from khosiness.sandbox.local import LocalSandbox


def python_command():
    return Path(sys.executable).name


def make_sandbox(tmp_path):
    return LocalSandbox(
        tmp_path,
        allowed_commands={python_command()},
    )


def test_shell_captures_stdout(tmp_path):
    sandbox = make_sandbox(tmp_path)

    result = sandbox.execute(
        ExecRequest(
            argv=[
                sys.executable,
                "-c",
                "print(123)",
            ],
            cwd=tmp_path,
        )
    )

    assert result.exit_code == 0
    assert result.stdout.strip() == "123"


def test_nonzero_exit_is_data_not_exception(tmp_path):
    sandbox = make_sandbox(tmp_path)

    result = sandbox.execute(
        ExecRequest(
            argv=[
                sys.executable,
                "-c",
                "raise SystemExit(7)",
            ],
            cwd=tmp_path,
        )
    )

    assert result.exit_code == 7
    assert result.timed_out is False


def test_timeout_is_reported(tmp_path):
    sandbox = make_sandbox(tmp_path)

    result = sandbox.execute(
        ExecRequest(
            argv=[
                sys.executable,
                "-c",
                "import time; time.sleep(2)",
            ],
            cwd=tmp_path,
            timeout_seconds=1,
        )
    )

    assert result.timed_out is True
    assert result.exit_code is None


def test_disallowed_command_is_rejected(tmp_path):
    sandbox = make_sandbox(tmp_path)

    try:
        sandbox.execute(
            ExecRequest(
                argv=["definitely-not-allowed"],
                cwd=tmp_path,
            )
        )
        assert False, "expected command rejection"
    except PermissionError:
        pass


def test_cwd_cannot_escape_workspace(tmp_path):
    sandbox = make_sandbox(tmp_path)

    try:
        sandbox.execute(
            ExecRequest(
                argv=[
                    sys.executable,
                    "-c",
                    "print(1)",
                ],
                cwd=tmp_path.parent,
            )
        )
        assert False, "expected cwd rejection"
    except PermissionError:
        pass
