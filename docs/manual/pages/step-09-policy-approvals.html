from dataclasses import dataclass

from .risk import Decision, RiskLevel


DEFAULT_POLICY = {
    RiskLevel.READ_ONLY: Decision.ALLOW,
    RiskLevel.PROCESS_EXECUTION: Decision.ALLOW,
    RiskLevel.WORKSPACE_WRITE: Decision.ASK,
    RiskLevel.NETWORK: Decision.ASK,
    RiskLevel.SECRET_ACCESS: Decision.DENY,
    RiskLevel.DESTRUCTIVE: Decision.DENY,
}


@dataclass(frozen=True)
class Action:
    name: str
    risk: RiskLevel
    arguments: dict


class PolicyEngine:
    def __init__(
        self,
        policy: dict[RiskLevel, Decision] | None = None,
    ):
        self.policy = policy or DEFAULT_POLICY

    def evaluate(self, action: Action) -> Decision:
        return self.policy.get(
            action.risk,
            Decision.DENY,
        )

from collections.abc import Callable

from .engine import Action


class ApprovalUI:
    def __init__(
        self,
        input_fn: Callable[[str], str] = input,
    ):
        self.input_fn = input_fn

    def request(self, action: Action) -> bool:
        prompt = (
            "\nAgent requests an action requiring approval:\n"
            f"  action: {action.name}\n"
            f"  risk: {action.risk.value}\n"
            f"  arguments: {action.arguments}\n"
            "Approve? [y/N] "
        )
        answer = self.input_fn(prompt)
        return answer.strip().lower() in {
            "y",
            "yes",
        }

from khosiness.policy.approval import ApprovalUI
from khosiness.policy.engine import Action, PolicyEngine
from khosiness.policy.risk import Decision, RiskLevel


def action(risk):
    return Action(
        name="example",
        risk=risk,
        arguments={"x": 1},
    )


def test_read_only_is_allowed():
    assert (
        PolicyEngine().evaluate(
            action(RiskLevel.READ_ONLY)
        )
        is Decision.ALLOW
    )


def test_workspace_write_requires_approval():
    assert (
        PolicyEngine().evaluate(
            action(RiskLevel.WORKSPACE_WRITE)
        )
        is Decision.ASK
    )


def test_destructive_is_denied():
    assert (
        PolicyEngine().evaluate(
            action(RiskLevel.DESTRUCTIVE)
        )
        is Decision.DENY
    )


def test_unknown_policy_entry_defaults_to_deny():
    engine = PolicyEngine(
        {
            RiskLevel.READ_ONLY: Decision.ALLOW,
        }
    )

    assert (
        engine.evaluate(
            action(RiskLevel.NETWORK)
        )
        is Decision.DENY
    )


def test_approval_ui_accepts_yes():
    ui = ApprovalUI(input_fn=lambda _: "yes")
    assert ui.request(
        action(RiskLevel.WORKSPACE_WRITE)
    ) is True


def test_approval_ui_defaults_to_no():
    ui = ApprovalUI(input_fn=lambda _: "")
    assert ui.request(
        action(RiskLevel.WORKSPACE_WRITE)
    ) is False
