What this step does

What it builds on

The loop owns current state, the event store records facts, and policy controls actions. Those pieces need a recovery path.

Goal

Reconstruct a stopped run from durable evidence and continue safely from a known checkpoint.

import os
from pathlib import Path

from .state import RunState


def save_checkpoint(
    state: RunState,
    path: Path,
) -> None:
    """Atomically replace the committed state snapshot."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(
        path.suffix + ".tmp"
    )

    temporary.write_text(
        state.model_dump_json(indent=2),
        encoding="utf-8",
    )

    os.replace(temporary, path)


def load_checkpoint(path: Path) -> RunState:
    return RunState.model_validate_json(
        path.read_text(encoding="utf-8")
    )
src/khosiness/resume.py complete file 
from pathlib import Path

from .checkpoint import load_checkpoint
from .task import Task


def load_run(run_dir: Path) -> tuple[Task, object]:
    task = Task.model_validate_json(
        (run_dir / "task.json").read_text(
            encoding="utf-8"
        )
    )
    state = load_checkpoint(
        run_dir / "state.json"
    )
    return task, state


def resume_run(
    run_dir: Path,
    runner,
    **dependencies,
):
    task, state = load_run(run_dir)

    if state.finished:
        return state

    return runner(
        task=task,
        state=state,
        **dependencies,
    )

from khosiness.checkpoint import (
    load_checkpoint,
    save_checkpoint,
)
from khosiness.resume import resume_run
from khosiness.state import RunState
from khosiness.task import Task


def test_checkpoint_round_trip(tmp_path):
    path = tmp_path / "state.json"
    state = RunState(
        run_id="r1",
        step=4,
        observations=[{"x": 1}],
    )

    save_checkpoint(state, path)
    restored = load_checkpoint(path)

    assert restored == state


def test_checkpoint_replaces_previous_state(tmp_path):
    path = tmp_path / "state.json"

    save_checkpoint(
        RunState(run_id="r1", step=1),
        path,
    )
    save_checkpoint(
        RunState(run_id="r1", step=2),
        path,
    )

    assert load_checkpoint(path).step == 2


def test_resume_does_not_rerun_finished_state(tmp_path):
    run_dir = tmp_path / "run"
    run_dir.mkdir()

    task = Task(
        id="t1",
        objective="x",
        workspace=tmp_path,
    )
    (run_dir / "task.json").write_text(
        task.model_dump_json(),
        encoding="utf-8",
    )
    save_checkpoint(
        RunState(
            run_id="r1",
            finished=True,
            final_answer="done",
        ),
        run_dir / "state.json",
    )

    called = False

    def runner(**kwargs):
        nonlocal called
        called = True

    state = resume_run(
        run_dir,
        runner,
    )

    assert called is False
    assert state.final_answer == "done"


def test_resume_passes_unfinished_state_to_runner(tmp_path):
    run_dir = tmp_path / "run"
    run_dir.mkdir()

    task = Task(
        id="t1",
        objective="x",
        workspace=tmp_path,
    )
    (run_dir / "task.json").write_text(
        task.model_dump_json(),
        encoding="utf-8",
    )
    save_checkpoint(
        RunState(
            run_id="r1",
            step=3,
        ),
        run_dir / "state.json",
    )

    def runner(*, task, state, **kwargs):
        state.step += 1
        return state

    state = resume_run(
        run_dir,
        runner,
    )

    assert state.step == 4
