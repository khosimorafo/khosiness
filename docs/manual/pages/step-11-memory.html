What this step does

Store explicit cross-run knowledge and retrieve it with provenance and bounded relevance.

What it builds on

Step 10 handles the current run's continuity; it does not make reusable knowledge from earlier runs available.

Goal

Remember useful facts across runs without confusing memory with state, event history, or current model context.

End state

Memory schemas, storage, retrieval, and tests make the source and selection of remembered information inspectable.

Conceptual objective

Add cross-run knowledge using explicit memory types and bounded retrieval.

This stage adds continuity and modular expertise. Keep storage, retrieval and context injection separate so that long-lived knowledge does not silently become prompt clutter.

from datetime import datetime, timezone
from typing import Literal
from uuid import uuid4

from pydantic import BaseModel, Field


MemoryType = Literal[
    "episodic",
    "semantic",
    "procedural",
]


class Memory(BaseModel):
    id: str = Field(
        default_factory=lambda: str(uuid4())
    )
    memory_type: MemoryType
    content: str
    source_run_id: str | None = None
    created_at: datetime = Field(
        default_factory=lambda: datetime.now(
            timezone.utc
        )
    )
    updated_at: datetime = Field(
        default_factory=lambda: datetime.now(
            timezone.utc
        )
    )
    status: str = "active"
src/khosiness/memory/store.py complete file 
import sqlite3
from pathlib import Path

from .schema import Memory


SCHEMA = """
CREATE TABLE IF NOT EXISTS memories (
    id TEXT PRIMARY KEY,
    memory_type TEXT NOT NULL,
    content TEXT NOT NULL,
    source_run_id TEXT,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    status TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_memories_type
ON memories(memory_type);
"""


class SqliteMemoryStore:
    def __init__(self, path: Path):
        self.path = path
        self.path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )
        with self._connect() as connection:
            connection.executescript(SCHEMA)

    def _connect(self):
        return sqlite3.connect(self.path)

    def add(self, memory: Memory) -> None:
        with self._connect() as connection:
            connection.execute(
                """
                INSERT INTO memories (
                    id,
                    memory_type,
                    content,
                    source_run_id,
                    created_at,
                    updated_at,
                    status
                ) VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    memory.id,
                    memory.memory_type,
                    memory.content,
                    memory.source_run_id,
                    memory.created_at.isoformat(),
                    memory.updated_at.isoformat(),
                    memory.status,
                ),
            )

    def set_status(
        self,
        memory_id: str,
        status: str,
    ) -> None:
        with self._connect() as connection:
            connection.execute(
                """
                UPDATE memories
                SET status = ?
                WHERE id = ?
                """,
                (status, memory_id),
            )

    def search(
        self,
        query: str,
        memory_types: list[str] | None = None,
        limit: int = 5,
    ) -> list[Memory]:
        clauses = [
            "status = 'active'",
            "content LIKE ?",
        ]
        parameters: list[object] = [
            f"%{query}%"
        ]

        if memory_types:
            placeholders = ",".join(
                "?" for _ in memory_types
            )
            clauses.append(
                f"memory_type IN ({placeholders})"
            )
            parameters.extend(memory_types)

        parameters.append(limit)

        sql = f"""
            SELECT
                id,
                memory_type,
                content,
                source_run_id,
                created_at,
                updated_at,
                status
            FROM memories
            WHERE {" AND ".join(clauses)}
            ORDER BY updated_at DESC
            LIMIT ?
        """

        with self._connect() as connection:
            rows = connection.execute(
                sql,
                parameters,
            ).fetchall()

        return [
            Memory(
                id=row[0],
                memory_type=row[1],
                content=row[2],
                source_run_id=row[3],
                created_at=row[4],
                updated_at=row[5],
                status=row[6],
            )
            for row in rows
        ]
src/khosiness/memory/retrieval.py complete file 
from .store import SqliteMemoryStore


class MemoryRetriever:
    def __init__(
        self,
        store: SqliteMemoryStore,
        limit: int = 5,
    ):
        self.store = store
        self.limit = limit

    def retrieve(
        self,
        query: str,
    ) -> list[str]:
        memories = self.store.search(
            query=query,
            limit=self.limit,
        )

        return [
            (
                f"[{memory.memory_type}] "
                f"{memory.content} "
                f"(source_run={memory.source_run_id})"
            )
            for memory in memories
        ]
tests/test_memory.py complete file 
from khosiness.memory.retrieval import MemoryRetriever
from khosiness.memory.schema import Memory
from khosiness.memory.store import SqliteMemoryStore


def test_memory_survives_new_store_instance(tmp_path):
    path = tmp_path / "memory.db"

    first = SqliteMemoryStore(path)
    first.add(
        Memory(
            memory_type="semantic",
            content="Database retries live in db/config.py",
            source_run_id="run-a",
        )
    )

    second = SqliteMemoryStore(path)
    results = second.search("retries")

    assert len(results) == 1
    assert results[0].source_run_id == "run-a"


def test_inactive_memory_is_not_retrieved(tmp_path):
    store = SqliteMemoryStore(
        tmp_path / "memory.db"
    )
    memory = Memory(
        memory_type="semantic",
        content="Old configuration fact",
    )
    store.add(memory)
    store.set_status(memory.id, "inactive")

    assert store.search("configuration") == []


def test_memory_type_filter(tmp_path):
    store = SqliteMemoryStore(
        tmp_path / "memory.db"
    )
    store.add(
        Memory(
            memory_type="semantic",
            content="run tests before release",
        )
    )
    store.add(
        Memory(
            memory_type="procedural",
            content="run tests before release",
        )
    )

    results = store.search(
        "tests",
        memory_types=["procedural"],
    )

    assert len(results) == 1
    assert results[0].memory_type == "procedural"


def test_retriever_returns_bounded_rendering(tmp_path):
    store = SqliteMemoryStore(
        tmp_path / "memory.db"
    )

    for index in range(10):
        store.add(
            Memory(
                memory_type="semantic",
                content=f"retry fact {index}",
            )
        )

    rendered = MemoryRetriever(
        store,
        limit=3,
    ).retrieve("retry")

    assert len(rendered) == 3
