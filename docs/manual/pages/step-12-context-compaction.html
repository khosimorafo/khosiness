What this step does

Compact active context when it grows, while retaining original history and work that remains unresolved.

What it builds on

The context builder selects messages, and memory and event history provide durable material; long runs still need a size boundary.

Goal

Keep future model calls useful and bounded without silently discarding important obligations or raw records.

End state

A compaction mechanism and tests show what is summarized, what remains active, and what stays recoverable.

Conceptual objective

Bound active context while preserving raw history and unresolved work.

This stage adds continuity and modular expertise. Keep storage, retrieval and context injection separate so that long-lived knowledge does not silently become prompt clutter.

from pydantic import BaseModel, Field

from khosiness.state import RunState


class Compaction(BaseModel):
    through_step: int
    summary: str
    decisions: list[str] = Field(
        default_factory=list
    )
    evidence: list[str] = Field(
        default_factory=list
    )
    unresolved: list[str] = Field(
        default_factory=list
    )


def should_compact(
    *,
    context_characters: int,
    threshold: int = 60_000,
) -> bool:
    return context_characters > threshold


def deterministic_compact(
    state: RunState,
    keep_recent: int = 6,
) -> tuple[Compaction, list[dict]]:
    """A deliberately simple compactor for learning.

    It preserves recent observations verbatim and records how
    many older observations were summarized. Later you can
    replace the summary strategy with a model-assisted one.
    """

    if keep_recent < 1:
        raise ValueError(
            "keep_recent must be >= 1"
        )

    old = state.observations[:-keep_recent]
    recent = state.observations[-keep_recent:]

    compaction = Compaction(
        through_step=state.step,
        summary=(
            f"{len(old)} older observations "
            "were compacted. Inspect the event log "
            "for the original evidence."
        ),
        decisions=list(state.completed_work),
        evidence=[
            observation["tool"]
            for observation in old
            if "tool" in observation
        ],
        unresolved=list(state.pending_actions),
    )

    return compaction, recent
tests/test_compaction.py complete file 
import pytest

from khosiness.context.compaction import (
    deterministic_compact,
    should_compact,
)
from khosiness.state import RunState


def test_compaction_threshold():
    assert should_compact(
        context_characters=60_001,
        threshold=60_000,
    )
    assert not should_compact(
        context_characters=60_000,
        threshold=60_000,
    )


def test_compactor_preserves_recent_observations():
    state = RunState(
        run_id="r1",
        step=30,
        observations=[
            {"tool": f"tool-{index}"}
            for index in range(10)
        ],
    )

    compacted, recent = deterministic_compact(
        state,
        keep_recent=3,
    )

    assert len(recent) == 3
    assert recent[0]["tool"] == "tool-7"
    assert "7 older observations" in compacted.summary


def test_compactor_carries_decisions_and_pending_work():
    state = RunState(
        run_id="r1",
        completed_work=["selected config.py"],
        pending_actions=[{"action": "run tests"}],
        observations=[{"tool": "search"}],
    )

    compacted, _ = deterministic_compact(
        state,
        keep_recent=1,
    )

    assert compacted.decisions == [
        "selected config.py"
    ]
    assert compacted.unresolved == [
        {"action": "run tests"}
    ]


def test_keep_recent_must_be_positive():
    with pytest.raises(ValueError):
        deterministic_compact(
            RunState(run_id="r1"),
            keep_recent=0,
        )
