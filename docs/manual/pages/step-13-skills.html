from pathlib import Path

from .catalog import SkillSummary


def parse_front_matter(
    text: str,
) -> tuple[dict[str, str], str]:
    lines = text.splitlines()

    if not lines or lines[0].strip() != "---":
        raise ValueError(
            "skill must start with YAML-like front matter"
        )

    metadata: dict[str, str] = {}
    closing_index = None

    for index, line in enumerate(
        lines[1:],
        start=1,
    ):
        if line.strip() == "---":
            closing_index = index
            break

        if ":" not in line:
            raise ValueError(
                f"invalid metadata line: {line}"
            )

        key, value = line.split(":", 1)
        metadata[key.strip()] = value.strip()

    if closing_index is None:
        raise ValueError(
            "front matter is not closed"
        )

    body = "\n".join(
        lines[closing_index + 1:]
    ).strip()

    return metadata, body


class SkillLoader:
    def __init__(self, skills_dir: Path):
        self.skills_dir = skills_dir.resolve()

    def catalog(self) -> list[SkillSummary]:
        summaries = []

        for path in sorted(
            self.skills_dir.glob("*/SKILL.md")
        ):
            metadata, _ = parse_front_matter(
                path.read_text(encoding="utf-8")
            )
            summaries.append(
                SkillSummary(
                    name=metadata["name"],
                    description=metadata[
                        "description"
                    ],
                    path=path,
                )
            )

        return summaries

    def load(self, name: str) -> str:
        for summary in self.catalog():
            if summary.name == name:
                _, body = parse_front_matter(
                    summary.path.read_text(
                        encoding="utf-8"
                    )
                )
                return body

        raise KeyError(f"unknown skill: {name}")

import pytest

from khosiness.skills.loader import SkillLoader


def make_skill(
    root,
    name="example",
    description="Example skill",
    body="# Procedure\nDo the thing.",
):
    folder = root / name
    folder.mkdir(parents=True)
    (folder / "SKILL.md").write_text(
        f"---\n"
        f"name: {name}\n"
        f"description: {description}\n"
        f"---\n\n"
        f"{body}\n",
        encoding="utf-8",
    )


def test_catalog_exposes_metadata_not_full_body(tmp_path):
    make_skill(tmp_path)
    loader = SkillLoader(tmp_path)

    catalog = loader.catalog()

    assert catalog[0].name == "example"
    assert catalog[0].description == "Example skill"
    assert "Do the thing" not in repr(catalog[0])


def test_load_returns_full_body(tmp_path):
    make_skill(tmp_path)
    loader = SkillLoader(tmp_path)

    body = loader.load("example")

    assert "# Procedure" in body
    assert "Do the thing" in body


def test_unknown_skill_fails_cleanly(tmp_path):
    loader = SkillLoader(tmp_path)

    with pytest.raises(
        KeyError,
        match="unknown skill",
    ):
        loader.load("missing")
