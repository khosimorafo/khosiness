from pydantic import BaseModel


class EvalResult(BaseModel):
    task_id: str
    success: bool
    steps: int
    tool_calls: int
    failed_tool_calls: int
    input_tokens: int | None = None
    output_tokens: int | None = None
    wall_ms: int
    human_interventions: int = 0
    failure_class: str | None = None
    run_id: str


class EvalSummary(BaseModel):
    tasks: int
    successes: int
    success_rate: float
    total_steps: int
    total_tool_calls: int


def summarize(
    results: list[EvalResult],
) -> EvalSummary:
    tasks = len(results)
    successes = sum(
        1 for result in results
        if result.success
    )

    return EvalSummary(
        tasks=tasks,
        successes=successes,
        success_rate=(
            successes / tasks
            if tasks
            else 0.0
        ),
        total_steps=sum(
            result.steps for result in results
        ),
        total_tool_calls=sum(
            result.tool_calls
            for result in results
        ),
    )
src/khosiness/evals/runner.py complete file 
from collections.abc import Callable
from time import perf_counter

from .metrics import EvalResult


def run_eval(
    *,
    task_id: str,
    execute: Callable[[], object],
    grade: Callable[[object], bool],
    metrics_from_run: Callable[
        [object],
        dict,
    ],
) -> EvalResult:
    started = perf_counter()
    run = execute()
    success = grade(run)
    metrics = metrics_from_run(run)

    return EvalResult(
        task_id=task_id,
        success=success,
        steps=metrics["steps"],
        tool_calls=metrics["tool_calls"],
        failed_tool_calls=metrics[
            "failed_tool_calls"
        ],
        input_tokens=metrics.get(
            "input_tokens"
        ),
        output_tokens=metrics.get(
            "output_tokens"
        ),
        wall_ms=int(
            (perf_counter() - started) * 1000
        ),
        human_interventions=metrics.get(
            "human_interventions",
            0,
        ),
        failure_class=metrics.get(
            "failure_class"
        ),
        run_id=metrics["run_id"],
    )

from types import SimpleNamespace

from khosiness.evals.metrics import (
    EvalResult,
    summarize,
)
from khosiness.evals.runner import run_eval


def test_summary_calculates_success_rate():
    results = [
        EvalResult(
            task_id="a",
            success=True,
            steps=2,
            tool_calls=3,
            failed_tool_calls=0,
            wall_ms=10,
            run_id="r1",
        ),
        EvalResult(
            task_id="b",
            success=False,
            steps=4,
            tool_calls=5,
            failed_tool_calls=1,
            wall_ms=20,
            run_id="r2",
            failure_class="missing_context",
        ),
    ]

    summary = summarize(results)

    assert summary.tasks == 2
    assert summary.successes == 1
    assert summary.success_rate == 0.5
    assert summary.total_steps == 6


def test_eval_runner_combines_execution_grading_and_metrics():
    def execute():
        return SimpleNamespace(
            answer="DB_RETRY_COUNT = 3",
        )

    def grade(run):
        return "DB_RETRY_COUNT" in run.answer

    def metrics(run):
        return {
            "steps": 2,
            "tool_calls": 1,
            "failed_tool_calls": 0,
            "run_id": "r1",
        }

    result = run_eval(
        task_id="retry-config",
        execute=execute,
        grade=grade,
        metrics_from_run=metrics,
    )

    assert result.success is True
    assert result.steps == 2
    assert result.run_id == "r1"


def test_empty_summary_is_defined():
    summary = summarize([])
    assert summary.success_rate == 0.0
